window.runMachineToggle = function(showId, hideId, element) {
    document.getElementById(hideId).style.display = 'none';
    document.getElementById(showId).style.display = 'block';
    
    document.getElementById('tab-v').style.color = '#666';
    document.getElementById('tab-v').style.borderBottom = '3px solid transparent';
    document.getElementById('tab-e').style.color = '#666';
    document.getElementById('tab-e').style.borderBottom = '3px solid transparent';
    
    element.style.color = '#0073aa';
    element.style.borderBottom = '3px solid #0073aa';
};

window.processIncomingFile = function(input) {
    var file = input.files[0];
    if (!file) return;

    document.getElementById('file-status').innerText = "✔️ تم تحميل: " + file.name;
    var reader = new FileReader();

    if (file.type.match('image.*')) {
        reader.onload = function(e) {
            var imgEl = document.getElementById('image-render');
            imgEl.src = e.target.result;
            imgEl.style.display = 'block';
            document.getElementById('text-render').innerHTML = "<strong>🖼️ تم استيراد ومعاينة الصورة بنجاح:</strong> " + file.name;
            updateQRCodeData("Image File: " + file.name);
            window.runMachineToggle('view-pane', 'edit-pane', document.getElementById('tab-v'));
        };
        reader.readAsDataURL(file);
    } else {
        reader.onload = function(e) {
            document.getElementById('text-input').value = e.target.result;
            document.getElementById('image-render').style.display = 'none';
            document.getElementById('text-render').innerHTML = "<strong>📄 تم قراءة المستند وجلبه للمحرر:</strong> " + file.name;
            updateQRCodeData(e.target.result);
            window.runMachineToggle('view-pane', 'edit-pane', document.getElementById('tab-v'));
        };
        reader.readAsText(file);
    }
};

function updateQRCodeData(dataString) {
    var qrContainer = document.getElementById("qrcode");
    qrContainer.innerHTML = ""; 
    
    if(!dataString || dataString.trim() === "") {
        document.getElementById('qr-section').style.display = 'none';
        return;
    }
    
    new QRCode(qrContainer, {
        text: dataString,
        width: 140,
        height: 140,
        colorDark : "#000000",
        colorLight : "#ffffff",
        correctLevel : QRCode.CorrectLevel.H
    });
    document.getElementById('qr-section').style.display = 'block';
}
