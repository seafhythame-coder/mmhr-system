<!-- برمجية التحديث الفوري المباشر بدون أزرار -->
<script>
function generateFastQR() {
    var rawData = document.getElementById('direct-qr-url').value;
    var qrHolder = document.getElementById('direct-qr-holder');
    var emptyMsg = document.getElementById('direct-empty-msg');
    var qrImg = document.getElementById('google-qr-img');
    
    if (rawData.trim() === "") {
        qrHolder.style.display = 'none';
        emptyMsg.style.display = 'block';
        return;
    }
    
    // استخدام محرك جوجل الفوري لتوليد وتظبيط الـ QR كصورة عادية مستحيل تفتح PDF
    var googleQRUrl = "https://googleapis.com" + encodeURIComponent(rawData) + "&choe=UTF-8";
    qrImg.src = googleQRUrl;
    
    qrHolder.style.display = 'block';
    emptyMsg.style.display = 'none';
}

function updateBankCard() {
    var bName = document.getElementById('direct-bname').value;
    var bIban = document.getElementById('direct-biban').value;
    var card = document.getElementById('direct-bank-render');
    
    if (bName.trim() !== "" || bIban.trim() !== "") {
        document.getElementById('lbl-bname').innerText = bName || '-';
        document.getElementById('lbl-biban').innerText = bIban || '-';
        card.style.display = 'block';
    } else {
        card.style.display = 'none';
    }
}
</script>
