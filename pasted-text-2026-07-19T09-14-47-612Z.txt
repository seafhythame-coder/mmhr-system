window.processIncomingFileDirect = function(input) {
    var file = input.files[0];
    if (!file) return;

    document.getElementById('file-status').innerText = "✔️ تم تحميل الملف: " + file.name;
    var reader = new FileReader();

    if (file.type.match('image.*')) {
        reader.onload = function(e) {
            var imgEl = document.getElementById('image-render');
            imgEl.src = e.target.result;
            imgEl.style.display = 'block';
            document.getElementById('text-render').innerHTML = "<strong>🖼️ تم استيراد ومعاينة الصورة المرفوعة بنجاح:</strong> " + file.name;
            updateQRCodeDataDirect("Image File: " + file.name);
        };
        reader.readAsDataURL(file);
    } else {
        reader.onload = function(e) {
            document.getElementById('text-input').value = e.target.result;
            document.getElementById('image-render').style.display = 'none';
            document.getElementById('text-render').innerHTML = "<strong>📄 تم قراءة المستند وجلبه للمحرر بنجاح:</strong> " + file.name;
            updateQRCodeDataDirect(e.target.result);
        };
        reader.readAsText(file);
    }
};

function updateQRCodeDataDirect(dataString) {
    var qrContainer = document.getElementById("qrcode");
    qrContainer.innerHTML = ""; 
    
    if(!dataString || dataString.trim() === "") {
        document.getElementById('qr-section').style.display = 'none';
        return;
    }
    
    new QRCode(qrContainer, {
        text: dataString,
        width: 150,
        height: 150,
        colorDark : "#000000",
        colorLight : "#ffffff",
        correctLevel : QRCode.CorrectLevel.H
    });
    document.getElementById('qr-section').style.display = 'block';
}
